# Development and test dependencies. NOT needed to run the application.
#
#   pip install -r requirements.txt -r requirements-dev.txt
#
# The canonical test runner remains `python run_all_tests.py` — it drives all
# 36 entries as subprocesses and forces UTF-8 on the children, which matters:
# a test that printed "mm³" once died with UnicodeEncodeError on a cp1256
# console AFTER every assertion in it had passed, reporting a green file as a
# failure. pytest runs alongside it and both must pass; it is not a
# replacement.
pytest

# SAST gate. Run as a CI job, never as a pre-commit hook — see .semgrep.yml for
# why. Installing it also pulls in the opentelemetry SDK as a transitive
# dependency, which is how telemetry.py's OTel bridge comes to be exercised
# locally; telemetry.py works identically without it.
semgrep
